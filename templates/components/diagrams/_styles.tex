% ============================================================================
% BOX STYLES
% ============================================================================

\tikzset{
    % Standard box (light theme)
    diagram box/.style={
        rectangle, rounded corners=4pt,
        draw=diagramBorder, fill=diagramSurface,
        minimum width=2.5cm, minimum height=1cm,
        align=center, font=\small\bfseries,
        text=diagramTextDark
    },
    % Primary emphasis box
    diagram box primary/.style={
        diagram box,
        draw=diagramPrimary, fill=diagramPrimary!15,
        text=diagramPrimary
    },
    % Secondary box
    diagram box secondary/.style={
        diagram box,
        draw=diagramSecondary, fill=diagramSecondary!10
    },
    % Success/positive box
    diagram box success/.style={
        diagram box,
        draw=diagramSuccess, fill=diagramSuccess!10,
        text=diagramSuccess
    },
    % Warning box
    diagram box warning/.style={
        diagram box,
        draw=diagramWarning, fill=diagramWarning!10
    },
    % Error/negative box
    diagram box error/.style={
        diagram box,
        draw=diagramError, fill=diagramError!10,
        text=diagramError
    },
    % Dark theme box
    diagram box dark/.style={
        rectangle, rounded corners=4pt,
        draw=diagramDarkBorder, fill=diagramDarkBg,
        minimum width=2.5cm, minimum height=1cm,
        align=center, font=\small\bfseries,
        text=diagramDarkText
    },
    % Dark theme primary
    diagram box dark primary/.style={
        diagram box dark,
        draw=diagramDarkPrimary, fill=diagramDarkSurface

}
